# Lore v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a Lore mode for multi-campaign markdown pages (tree + tags) in `localStorage`, with Pin to Board as a linked `lore` card, portable-save inclusion, on static GitHub Pages / local-only.

**Architecture:** New `LORE` IIFE + `#lore` full-bleed shell (mirror Forge/Board). Store `bg.lore.v1` with campaigns → pages (`parentId`, `tags`, `body`). Board gains `lore` card type + `BOARD.addLorePage` / `BOARD.mdHtml`. Script 1 portable save merges lore by campaign/page id. No server APIs.

**Tech Stack:** Vanilla JS in `app.template.html`; `localStorage`; existing BOARD `md()` markdown; `python3 build_bundles.py`. No unit test suite — verify with `node --check`, `build_bundles.py --check`, browser smoke.

## Global Constraints

- Local-only, static GitHub Pages — no server, no accounts, no File System Access API.
- Edit `app.template.html` only for app code; rebuild `index.html` via `python3 build_bundles.py`.
- Linked Board cards store `{type:"lore", campaignId, pageId}` — never embed page body.
- Missing page/campaign stays valid in `vCard`; render honest missing notice.
- Guard cross-module calls: `window.LORE`, `window.BOARD`, `window.TRK`.
- Destructive deletes use `TRK.confirmSwap`, never `window.confirm`.
- Do not put `lore` on the Board add rail.
- Capture-from-Board (Phase 4), excerpt pin, wiki backlinks are out of scope.
- Keep docs in the shipping change: PRODUCT, DESIGN, CLAUDE, workflow-spec, plan note.

- [ ] **Step 5: LORE IIFE — validators + persist + minimal setActive**

Append a new `<script>` after FORGE (before `</body>`):

```js
const LORE = (() => {
  const STORE = "bg.lore.v1";
  const MAX_CAMPAIGNS = 20;
  const MAX_PAGES = 200;
  const MAX_TAGS = 24;
  const MD_SOFT = 20000, MD_WARN = 60000, MD_HARD = 120000;
  const esc = s => String(s==null?"":s).replace(/[&<>"]/g, c =>
    ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));
  const uid = () => "l" + Math.random().toString(36).slice(2, 10);
  function vStr(v, max, fb){
    if (typeof v !== "string") return fb;
    return v.length > max ? v.slice(0, max) : v;
  }
  function vPage(raw, knownIds){
    if (!raw || typeof raw !== "object") return null;
    const id = (typeof raw.id === "string" && raw.id) ? raw.id.slice(0, 64) : uid();
    let parentId = (typeof raw.parentId === "string" && raw.parentId) ? raw.parentId.slice(0, 64) : null;
    if (parentId && knownIds && !knownIds.has(parentId)) parentId = null;
    const tagsIn = Array.isArray(raw.tags) ? raw.tags : [];
    const tags = [];
    for (const t of tagsIn.slice(0, MAX_TAGS)){
      if (typeof t !== "string") continue;
      const s = t.trim().slice(0, 40);
      if (s && !tags.includes(s)) tags.push(s);
    }
    let body = vStr(raw.body, MD_HARD, "");
    return {
      id, title: vStr((raw.title || "Untitled").trim() || "Untitled", 80, "Untitled"),
      parentId, tags, body,
      updatedAt: (typeof raw.updatedAt === "number" && Number.isFinite(raw.updatedAt))
        ? raw.updatedAt : Date.now(),
    };
  }
  function vCampaign(raw){
    if (!raw || typeof raw !== "object") return null;
    if (typeof raw.id !== "string" || !raw.id) return null;
    const pagesIn = Array.isArray(raw.pages) ? raw.pages : [];
    const known = new Set();
    for (const p of pagesIn.slice(0, MAX_PAGES))
      if (p && typeof p.id === "string" && p.id) known.add(p.id.slice(0, 64));
    const pages = [];
    for (const p of pagesIn.slice(0, MAX_PAGES)){
      const vp = vPage(p, known);
      if (vp) pages.push(vp);
    }
    return {
      id: raw.id.slice(0, 64),
      title: vStr((typeof raw.title === "string" && raw.title.trim() ? raw.title.trim() : "Campaign"), 80, "Campaign"),
      updatedAt: (typeof raw.updatedAt === "number" && Number.isFinite(raw.updatedAt))
        ? raw.updatedAt : Date.now(),
      pages,
    };
  }
  function seedLore(){
    const cid = "camp-" + uid();
    return {
      activeCampaignId: cid,
      campaigns: [{ id: cid, title: "Campaign 1", updatedAt: Date.now(), pages: [] }],
    };
  }
  function vLoreState(raw){
    if (!raw || !Array.isArray(raw.campaigns) || !raw.campaigns.length) return seedLore();
    const campaigns = [];
    for (const c of raw.campaigns.slice(0, MAX_CAMPAIGNS)){
      const vc = vCampaign(c);
      if (vc) campaigns.push(vc);
    }
    if (!campaigns.length) return seedLore();
    let activeCampaignId = typeof raw.activeCampaignId === "string"
      ? raw.activeCampaignId : campaigns[0].id;
    if (!campaigns.some(c => c.id === activeCampaignId)) activeCampaignId = campaigns[0].id;
    return { activeCampaignId, campaigns };
  }
  function load(){
    try{
      const raw = localStorage.getItem(STORE);
      if (!raw) return seedLore();
      return vLoreState(JSON.parse(raw));
    }catch(_){ return seedLore(); }
  }
  function save(){
    try{
      localStorage.setItem(STORE, JSON.stringify(vLoreState(state)));
      return true;
    }catch(_){
      if (typeof announceLive === "function")
        announceLive("Could not save Lore — browser storage may be full.");
      return false;
    }
  }
  let state = load();
  let active = false;
  let selectedPageId = null;
  let previewing = false;

  function activeCamp(){
    return state.campaigns.find(c => c.id === state.activeCampaignId) || state.campaigns[0];
  }
  function getPage(campaignId, pageId){
    const c = state.campaigns.find(x => x.id === campaignId);
    if (!c) return null;
    return c.pages.find(p => p.id === pageId) || null;
  }
  function getExport(){ return JSON.parse(JSON.stringify(vLoreState(state))); }
  function applyUserSave(parts){
    if (!parts || !Array.isArray(parts.campaigns)) return { loreUpserts: 0, ok: true };
    const map = new Map(state.campaigns.map(c => [c.id, c]));
    let upserts = 0;
    for (const raw of parts.campaigns.slice(0, MAX_CAMPAIGNS)){
      const incoming = vCampaign(raw);
      if (!incoming) continue;
      const local = map.get(incoming.id);
      if (!local){ map.set(incoming.id, incoming); upserts++; continue; }
      const pmap = new Map(local.pages.map(p => [p.id, p]));
      for (const p of incoming.pages) if (!pmap.has(p.id)){ pmap.set(p.id, p); upserts++; }
      // incoming overwrites same-id pages (portable import semantics: merge ids, prefer incoming body for shared ids — match boards' replace-same-id)
      for (const p of incoming.pages) pmap.set(p.id, p);
      map.set(incoming.id, {
        ...incoming,
        title: incoming.title || local.title,
        pages: [...pmap.values()].slice(0, MAX_PAGES),
      });
    }
    state.campaigns = [...map.values()].slice(0, MAX_CAMPAIGNS);
    if (!state.campaigns.some(c => c.id === state.activeCampaignId))
      state.activeCampaignId = state.campaigns[0].id;
    const ok = save();
    if (active) renderAll();
    return { loreUpserts: upserts, ok };
  }

  function setActive(on){
    active = !!on;
    const el = document.getElementById("lore");
    if (el) el.hidden = !active;
    if (active) renderAll();
  }
  function renderAll(){ /* Task 2 fills this */ }

  // stub bind so mode toggle does not throw — Task 2 attaches real handlers
  function bind(){ /* Task 2 */ }
  bind();

  return {
    setActive, getExport, applyUserSave, getPage,
    _test: { vLoreState, vCampaign, vPage, seedLore },
  };
})();
window.LORE = LORE;
```

```js
function depthOf(page, byId){
  let d = 0, cur = page, guard = 0;
  while (cur && cur.parentId && byId.get(cur.parentId) && guard++ < 40){
    cur = byId.get(cur.parentId); d++;
  }
  return d;
}
function visiblePages(camp, q, tagFilter){
  const ql = (q || "").trim().toLowerCase();
  return camp.pages.filter(p => {
    if (tagFilter && !p.tags.includes(tagFilter)) return false;
    if (!ql) return true;
    return p.title.toLowerCase().includes(ql)
      || p.tags.some(t => t.toLowerCase().includes(ql))
      || (p.body || "").toLowerCase().includes(ql);
  });
}
function renderTree(){
  const nav = document.getElementById("loreTree");
  const camp = activeCamp();
  if (!nav || !camp) return;
  const q = document.getElementById("loreQ")?.value || "";
  const tagFilter = nav.dataset.tagFilter || "";
  const pages = visiblePages(camp, q, tagFilter || null);
  const byId = new Map(camp.pages.map(p => [p.id, p]));
  // stable tree order: parent-before-child via repeated passes
  const ordered = [];
  const placed = new Set();
  let guard = 0;
  while (ordered.length < pages.length && guard++ < pages.length + 5){
    for (const p of pages){
      if (placed.has(p.id)) continue;
      if (!p.parentId || placed.has(p.parentId) || !byId.has(p.parentId) || !pages.some(x => x.id === p.parentId)){
        ordered.push(p); placed.add(p.id);
      }
    }
  }
  nav.innerHTML = "";
  for (const p of ordered){
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "lore-tree-item" + (p.id === selectedPageId ? " on" : "");
    btn.style.paddingLeft = (8 + depthOf(p, byId) * 14) + "px";
    btn.textContent = p.title;
    btn.onclick = () => { selectedPageId = p.id; previewing = false; renderAll(); };
    nav.appendChild(btn);
  }
}
```

```js
function fillLore(body, card){
  body.classList.add("lore-body");
  const page = (window.LORE && LORE.getPage)
    ? LORE.getPage(card.campaignId, card.pageId) : null;
  if (!page){
    const miss = document.createElement("p");
    miss.className = "lore-missing";
    miss.textContent = "Lore page not found — it may have been removed from the campaign.";
    body.appendChild(miss);
    return;
  }
  const view = document.createElement("div");
  view.className = "md-view lore-view";
  view.innerHTML = mdHtml(page.body);
  body.appendChild(view);
}
function addLorePage(campaignId, pageId){
  if (typeof campaignId !== "string" || !campaignId || typeof pageId !== "string" || !pageId)
    return { ok: false, error: "No lore page id." };
  const b = active();
  if (b.cards.length >= MAX_BOARD_CARDS)
    return { ok: false, error: "Board is full (" + MAX_BOARD_CARDS + " cards)." };
  const page = (window.LORE && LORE.getPage) ? LORE.getPage(campaignId, pageId) : null;
  addCard("lore", {
    campaignId: campaignId.slice(0, 64),
    pageId: pageId.slice(0, 64),
    title: (page && page.title) ? page.title : "Lore",
  });
  return { ok: true, name: page ? page.title : pageId };
}
```
